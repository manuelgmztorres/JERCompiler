# --- Palabras reservadas de EVALUAR usadas fuera de contexto ---

FUN ENT fueraDeContexto() {
    SINO;
    CUANDO 1: IMP "fuera de EVALUAR";
    PRED: IMP "fuera de EVALUAR";
    RET 0;
}

}
