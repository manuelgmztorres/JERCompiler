# 1. Declaraciones Globales de Variables y Constantes
CONST DEC PI -> 3.14159;
CONST CAD SALUDO -> "Bienvenido al sistema JER";
CONST BOO MODO_DEBUG -> VERDADERO;

ENT contadorGlobal -> 0;
DEC factorMultiplicador -> 1.5;
CAD mensajeActual -> "Estado OK";
CAR letraInicial -> 'J';
BOO sistemaActivo -> FALSO;

# 2. Arreglos y Matrices Multidimensionales
ENT vector[5] -> [10, 20, 30, 40, 50];
DEC matriz2D[2][3] -> [[1.1, 1.2, 1.3], [2.1, 2.2, 2.3]];
CAD catalogo[2] -> ["Elemento A", "Elemento B"];

# 3. Funciones Matemáticas y Lógicas
FUN calcularPotenciaYRaiz(DEC base, DEC exponente) {
    DEC pot -> base ** exponente;
    DEC raizCuadrada -> base // 2.0;
    DEC residuo -> base % 2.0;
    DEC opAritmetica -> (pot + raizCuadrada - residuo) * 2.0 / factorMultiplicador;
    RET opAritmetica;
}

FUN evaluarCalificacion(ENT nota) {
    EVALUAR nota {
        CUANDO 10:
            IMP "Excelente";
            contadorGlobal++;
        CUANDO 9:
            IMP "Muy Bien";
            contadorGlobal +-> 1;
        CUANDO 8:
            IMP "Bien";
        CUANDO 7:
            IMP "Regular";
        PRED:
            IMP "Necesita Mejorar";
            contadorGlobal --> 1;
    }
}

FUN flujoControlYBucles(ENT limite) {
    ENT i -> 0;
    
    # Bucle MIENTRAS con operadores relacionales y lógicos
    MIENTRAS i < limite AND (sistemaActivo = FALSO OR contadorGlobal != 0) {
        vector[i] -> vector[i] * 2;
        i++;
    }
    
    # Bucle REPETIR
    REPETIR 3 {
        IMP "Iteración de repetición fija";
    }
    
    # Estructura SI / SINO con anidación y bloque
    SI (limite >= 10 AND NOT limite = 100) {
        IMP "Límite en rango medio";
        SI limite = 50 {
            TERMINAR;
        }
    } SINO {
        IMP "Límite fuera de rango";
    }
}

FUN entradaSalidaYAsignaciones() {
    ENT lectura -> 0;
    OBT lectura;
    
    lectura +-> 10;
    lectura --> 2;
    
    IMP SALUDO;
    IMP lectura;
}

FUN principal() {
    IMP SALUDO;
    flujoControlYBucles(5);
    evaluarCalificacion(9);
    DEC resultado -> calcularPotenciaYRaiz(4.0, 3.0);
    IMP resultado;
    entradaSalidaYAsignaciones();
    RET 0;
}
