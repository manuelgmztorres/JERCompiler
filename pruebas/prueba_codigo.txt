ENT num1 -> 0;
ENT num2 -> 0;
BOO bandera -> VERDADERO;
CONST CAD nombre -> "Horacio";
ENT vector[2] -> [1,2];

FUN suma(){
	IMP("Ingrese el primer numero a sumar: ");
	OBT num1;
	IMP("Ingrese el segundo numero a sumar: ");
	OBT num2;
	IMP("La suma es: ");
	IMP(num1+num2);
}

FUN capturaBandera(BOO bandera){
	SI(bandera=VERDADERO){
		IMP("La bandera no ha sido tomado");
	}SINO{
		IMP("La bandera ha sido tomada");
	}
}

FUN pruebaVector(ENT vector){
	ENT contador -> 0;
	REPETIR (ENT j -> 0; j < 5; j +-> 1){
		IMP(vector[contador]);
		contador+->1;
	}
}
