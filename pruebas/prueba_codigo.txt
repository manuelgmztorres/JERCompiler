DEC num1 -> 0;
ENT num2 -> 0;
BOO bandera -> VERDADERO;
ent resta;
CONST CAD nombre -> "Horacio";

FUN VACIO suma(){
	IMP("Ingrese el primer numero a sumar: ");
	OBT num1;
	IMP("Ingre se el segundo numero a sumar: ");
	OBT num2;
	IMP("La suma es: ");
	IMP(num1+num2);
	resta -> num2 - num1;
}

FUN VACIO capturaBandera(BOO bandera){
	SI(bandera=VERDADERO){
		IMP("La bandera no ha sido tomado");
	}SINO{
		IMP("La bandera ha sido tomada");
	}
}

FUN VACIO mostrarNombre(CAD nombre){
	IMP "Tu nombre es: " + nombre;
}
