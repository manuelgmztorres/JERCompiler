# Prueba de diagnósticos hiperespecíficos del compilador JER

# ===== Grupo 1: Instrucciones de E/S y control incompletas =====

FUN grupoUno() {
    # 1. IMP vacío (sin expresión)
    IMP ;

    # 2. OBT vacío (sin identificador)
    OBT ;

    # 3. RET vacío (sin expresión de retorno)
    RET ;
}

# ===== Grupo 2: Cabeceras de control de flujo vacías =====

FUN grupoDos() {
    # 4. SI sin condición
    SI { IMP "A"; }

    # 5. MIENTRAS sin condición
    MIENTRAS { IMP "B"; }

    # 6. REPETIR sin número de repeticiones
    REPETIR { IMP "C"; }
}

# ===== Grupo 3: Asignaciones incompletas =====

FUN grupoTres() {
    # 7. Asignación directa vacía
    ENT x -> ;

    # 8. Asignación compuesta vacía (incremento)
    ENT y -> 0;
    y +-> ;
}

# ===== Grupo 4: Operadores aritméticos colgados =====

FUN grupoCuatro() {
    # 9. Operador sin operando derecho
    ENT a -> 10 + ;

    # 10. Dos operadores consecutivos
    ENT b -> 10 + * 2;
}
