# ==========================================
# HACER MIENTRAS: casos validos y regresion de los bugs de cascada
# especificos de su cola sintactica 'MIENTRAS ( condicion ) ;'.
# PARTE 1 (validos): 0 errores.
# PARTE 2 (cascada): cada bloque reporta exactamente 1 error.
# ==========================================

# --- PARTE 1: casos validos ---
FUN VACIO simple() {
    ENT i -> 0;
    HACER {
        i++;
    } MIENTRAS (i < 5);
}

FUN VACIO condicionCompuesta(ENT x) {
    HACER {
        x --> 1;
    } MIENTRAS (x > 0 AND x != 3);
}

FUN VACIO anidadoConSi(ENT n) {
    HACER {
        SI n > 10 {
            n --> 5;
        } SINO {
            n--;
        }
    } MIENTRAS (n > 0);
}

# --- PARTE 2: regresion de cascada en la cola de HACER ---

# Falta el bloque '{' tras HACER; la cola MIENTRAS parece valida y se rescata como tal.
FUN VACIO hacerSinBloque(ENT i) {
    HACER
    MIENTRAS (i < 3);
}

# Falta MIENTRAS tras '}': un solo error, la sentencia siguiente no debe duplicarse.
FUN VACIO hacerSinMientras() {
    ENT i -> 0;
    HACER {
        i++;
    }
    IMP "siguiente";
}

# Falta '(' antes de la condicion.
FUN VACIO hacerSinParentesis() {
    ENT i -> 0;
    HACER {
        i++;
    } MIENTRAS i < 3;
}

# Falta la condicion (parentesis vacios).
FUN VACIO hacerSinCondicion() {
    ENT i -> 0;
    HACER {
        i++;
    } MIENTRAS ();
}

# Falta ')' de cierre.
FUN VACIO hacerSinCierreParen() {
    ENT i -> 0;
    HACER {
        i++;
    } MIENTRAS (i < 3;
}

# Falta ';' final: la sentencia siguiente no debe duplicarse.
FUN VACIO hacerSinPuntoYComa() {
    ENT i -> 0;
    HACER {
        i++;
    } MIENTRAS (i < 3)
    IMP "siguiente";
}
