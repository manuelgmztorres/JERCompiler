FUN _inicio
  [B1]  -> fin
        LLAMAR principal, 0
        RET

FUN principal
  [B2]  -> B3
        IMP "Numero de Iteraciones"
        iteraciones = 0
        OBT iteraciones
        actual = 0
        siguiente = 1
        IMP "______________________________"
        IMP actual
        IMP siguiente
        i = 0
  [B3]  -> B4, B5
  L1:   t1 = i <= iteraciones
        SALTO_FAL t1, L2
  [B4]  -> B3
        suma = actual + siguiente
        actual = siguiente
        siguiente = suma
        IMP siguiente
        i = i + 1
        SALTO L1
  [B5]  -> fin
  L2:   RET
