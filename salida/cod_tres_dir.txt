FUN _inicio
        LLAMAR _global, 0
        LLAMAR principal, 0
        RET

FUN _global
        PI = 3.14159
        SALUDO = "Bienvenido al sistema JER"
        MODO_DEBUG = VERDADERO
        contador = 0
        factor = 1.5
        mensaje = "Estado OK"
        letra = 'J'
        activo = FALSO
        DIM 5
        vector = RESERVAR ENT, 1
        vector[0] = 10
        vector[1] = 20
        vector[2] = 30
        vector[3] = 40
        vector[4] = 50
        DIM 2
        DIM 3
        matriz = RESERVAR DEC, 2
        t1 = matriz[0]
        t1[0] = 1.1
        t1[1] = 1.2
        t1[2] = 1.3
        t2 = matriz[1]
        t2[0] = 2.1
        t2[1] = 2.2
        t2[2] = 2.3
        RET

FUN factorial
        RECIBIR n
        t1 = n <= 1
        SALTO_FAL t1, L1
        RET 1
        SALTO L2
  L1:   t2 = n - 1
        PARAM t2
        t3 = LLAMAR factorial, 1
        t4 = n * t3
        RET t4
  L2:   RET

FUN calcular
        RECIBIR base
        RECIBIR exponente
        potencia = base ** exponente
        raiz = base // 2.0
        residuo = base % 2.0
        t1 = potencia + raiz
        t2 = t1 - residuo
        t3 = t2 * 2.0
        t4 = t3 / factor
        RET t4

FUN evaluarNota
        RECIBIR nota
        t1 = nota == 10
        SALTO_VER t1, L3
        t2 = nota == 9
        SALTO_VER t2, L4
        SALTO L5
  L3:   IMP "Excelente"
        contador = contador + 1
        SALTO L6
  L4:   IMP "Muy bien"
        contador = contador + 1
        SALTO L6
  L5:   IMP "Necesita mejorar"
        contador = contador - 1
  L6:   RET

FUN controlDeFlujo
        RECIBIR limite
        i = 0
  L7:   t1 = i < limite
        SALTO_FAL t1, L9
        t2 = activo == FALSO
        SALTO_VER t2, L8
        t3 = contador != 0
        t2 = t3
  L8:   t1 = t2
  L9:   SALTO_FAL t1, L11
        t4 = vector[i]
        t5 = t4 * 2
        vector[i] = t5
        t6 = vector[i]
        t7 = t6 > 90
        SALTO_FAL t7, L10
        SALTO L11
  L10:  i = i + 1
        SALTO L7
  L11:  j = 0
  L12:  t8 = j < 3
        SALTO_FAL t8, L13
        IMP "Repeticion fija"
        j = j + 1
        SALTO L12
  L13:  t9 = limite >= 10
        SALTO_FAL t9, L14
        t10 = limite == 100
        t11 = NOT t10
        t9 = t11
  L14:  SALTO_FAL t9, L16
        t12 = matriz[0]
        t12[1] = factor
        t13 = limite == 50
        SALTO_FAL t13, L15
        IMP "limite es 50"
  L15:  SALTO L17
  L16:  IMP "Fuera de rango"
  L17:  RET

FUN entradaSalida
        lectura = 0
        OBT lectura
        lectura = lectura + 10
        lectura = lectura - 2
        IMP SALUDO
        IMP lectura
        RET

FUN contarHastaCero
        RECIBIR inicio
        i = inicio
  L18:  IMP i
        i = i - 1
        t1 = i > 0
        SALTO_VER t1, L18
        RET

FUN clasificar
        RECIBIR nota
        t1 = nota > 90
        SALTO_FAL t1, L19
        IMP "excelente"
        SALTO L22
  L19:  t2 = nota > 70
        SALTO_FAL t2, L20
        IMP "bien"
        SALTO L22
  L20:  t3 = nota > 50
        SALTO_FAL t3, L21
        IMP "regular"
        SALTO L22
  L21:  IMP "reprobado"
  L22:  RET

FUN principal
        IMP SALUDO
        PARAM 5
        LLAMAR controlDeFlujo, 1
        PARAM 9
        LLAMAR evaluarNota, 1
        PARAM 4.0
        PARAM 3.0
        resultado = LLAMAR calcular, 2
        IMP resultado
        PARAM 6
        t1 = LLAMAR factorial, 1
        IMP t1
        LLAMAR entradaSalida, 0
        PARAM 3
        LLAMAR contarHastaCero, 1
        PARAM 80
        LLAMAR clasificar, 1
        RET
